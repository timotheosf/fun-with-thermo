\tikzset{
    spring/.style={
        line width=0.4,
        color=ThemeColor!7!black!80,
        decorate, % Avisa que a linha não será reta
        decoration={
            coil, % Tipo da decoração (mola)
            aspect=0.4, % Deixa a mola "achatada" parecendo 3D
            segment length=3, % Comprimento de cada espira
            amplitude=2,    % "Raio" ou grossura da mola
            pre length=1,     % Pedaço de reta antes de começar a enrolar
            post length=1     % Pedaço de reta antes de terminar
        }
    }
}

\tikzset{
    FixedSpring/.style={
        line width=0.4,
        color=ThemeColor!7!black!80,
        decorate,
        decoration={
            coil,
            aspect=0.4,
            amplitude=2pt,
            segment length={(#1 - 4)/13 * 1pt}, 
            pre length=2pt,
            post length=2pt
        }
    }
}
