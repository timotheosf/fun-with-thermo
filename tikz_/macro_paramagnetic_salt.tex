\newcommand{\spinUp}[1]{
    \draw[dashed, color=AccentColor] (#1,-2.5) -- (#1,2.5);
    \particle{#1,0}
    \draw [-{Triangle[length=0.5em, width=0.4em]}, very thick, color=MediumRed]
    (#1-1,-1*1.73) -- (#1+1,1*1.73);
}
\newcommand{\spinDown}[1]{
    \draw[dashed, color=AccentColor] (#1,-2.5) -- (#1,2.5);
    \particle{#1,0}
    \draw [-{Triangle[length=0.5em, width=0.4em]}, very thick, color=MediumRed]
    (#1+1,1*1.73) -- (#1-1,-1*1.73);
}
\newcommand{\Hfield}[1]{
    \draw [-{Triangle[length=0.5em, width=0.4em]}, very thick, color=MediumGreen]
    (#1,-2.5) -- (#1,2.5);
    \node[color=MediumGreen] at (#1-1,0) {$\overrightarrow{\mathbf{H}}$};
}
